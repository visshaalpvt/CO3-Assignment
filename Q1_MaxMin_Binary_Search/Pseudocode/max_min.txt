MAX_MIN(A, lo, hi)
  if lo == hi: return (A[lo], A[lo])
  if hi == lo + 1:
      compare A[lo], A[hi] once
      return (smaller, larger)
  mid = floor((lo + hi) / 2)
  (minL, maxL) = MAX_MIN(A, lo, mid)
  (minR, maxR) = MAX_MIN(A, mid+1, hi)
  return (min(minL,minR), max(maxL,maxR))
