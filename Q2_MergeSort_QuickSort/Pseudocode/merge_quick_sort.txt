MERGE_SORT(A)
  if length(A) <= 1: return A
  split A into L and R
  return MERGE(MERGE_SORT(L), MERGE_SORT(R))

QUICK_SORT(A, lo, hi)
  if lo >= hi: return
  pivot = randomized pivot from A[lo..hi]
  partition into <= pivot and > pivot; place pivot in final position p
  QUICK_SORT(A, lo, p-1)
  QUICK_SORT(A, p+1, hi)
